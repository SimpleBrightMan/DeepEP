cmake_minimum_required(VERSION 3.20)
project(megamoe_ds_ladder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
    OUTPUT_VARIABLE pybind11_dir OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${pybind11_dir}")
find_package(pybind11 CONFIG REQUIRED)
link_directories("${SHMEM_ROOT}/lib")
set(DEEPEP_MEGAMOE_TIMER ON)
if(NOT IS_DIRECTORY "${MEGAMOE_SOURCE}/experiments/megamoe")
    message(FATAL_ERROR "MEGAMOE_SOURCE must be a frozen source snapshot")
endif()
add_subdirectory("${MEGAMOE_SOURCE}/experiments/megamoe" megamoe)
set_target_properties(megamoe_ext PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE)
